cmake_minimum_required(VERSION 3.21)
project(Dino VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
option(DINO_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(DINO_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets)
add_library(dino_core src/core/runner.cpp src/core/sprites.cpp src/core/theme.cpp)
target_include_directories(dino_core PUBLIC src)
target_link_libraries(dino_core PUBLIC Qt6::Core Qt6::Gui)
add_library(dino_ui src/ui/sprite_renderer.cpp src/ui/theme_watcher.cpp src/ui/game_view.cpp src/ui/window.cpp)
target_link_libraries(dino_ui PUBLIC dino_core Qt6::Widgets)
add_executable(dino src/main.cpp src/cli.cpp)
target_link_libraries(dino PRIVATE dino_ui)
foreach(target dino_core dino_ui dino)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(runner_tests tests/runner_tests.cpp)
    target_link_libraries(runner_tests PRIVATE dino_core)
    add_test(NAME runner COMMAND runner_tests)
    add_executable(theme_tests tests/theme_tests.cpp)
    target_link_libraries(theme_tests PRIVATE dino_core Qt6::Test)
    add_test(NAME theme COMMAND theme_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE dino_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME cli COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:dino>)
    add_test(NAME install COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh ${CMAKE_CURRENT_SOURCE_DIR} $<TARGET_FILE_DIR:dino>)
endif()
include(GNUInstallDirs)
install(TARGETS dino RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES licenses/omarchy-dino-MIT.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/dino)
install(FILES assets/io.github.dino_runner.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.dino_runner.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
